cmake_minimum_required(VERSION 3.16)
project(WebmaniaCpp VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(Webmania 0.1.0 CONFIG REQUIRED)
find_package(Threads REQUIRED)
file(GLOB WEBMANIA_CPP_SOURCES CONFIGURE_DEPENDS src/generated_*.cpp)
add_library(webmania_cpp ${WEBMANIA_CPP_SOURCES} src/file_results.cpp)
add_library(Webmania::cpp ALIAS webmania_cpp)
set_target_properties(webmania_cpp PROPERTIES EXPORT_NAME cpp)
target_compile_features(webmania_cpp PUBLIC cxx_std_17)
target_include_directories(webmania_cpp PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(webmania_cpp PUBLIC Webmania::webmania Threads::Threads)
target_compile_options(webmania_cpp PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(webmania_cpp PROPERTIES POSITION_INDEPENDENT_CODE YES)
install(TARGETS webmania_cpp EXPORT WebmaniaCppTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/webmania_cpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE README.md dependency-lock.json DESTINATION ${CMAKE_INSTALL_DATADIR}/webmania-cpp)
configure_package_config_file(cmake/WebmaniaCppConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/WebmaniaCppConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WebmaniaCpp)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/WebmaniaCppConfigVersion.cmake" VERSION 0.1.0 COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/WebmaniaCppConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/WebmaniaCppConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WebmaniaCpp)
install(EXPORT WebmaniaCppTargets NAMESPACE Webmania:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WebmaniaCpp)
set(CPACK_PACKAGE_NAME webmania-cpp)
set(CPACK_PACKAGE_VERSION "0.1.0-alpha.6")
set(CPACK_SYSTEM_NAME "linux-arm64")
set(CPACK_GENERATOR TGZ)
include(CPack)
