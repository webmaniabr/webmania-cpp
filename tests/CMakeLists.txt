cmake_minimum_required(VERSION 3.16)
project(WebmaniaCppConsumer LANGUAGES CXX)
find_package(WebmaniaCpp CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(wire-aliases wire_aliases.cpp)
target_link_libraries(wire-aliases PRIVATE Webmania::cpp Threads::Threads)
target_compile_options(wire-aliases PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(matrix generated_matrix.cpp)
target_link_libraries(matrix PRIVATE Webmania::cpp Threads::Threads)
target_compile_options(matrix PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(unit unit.cpp)
target_link_libraries(unit PRIVATE Webmania::cpp Threads::Threads)
target_compile_options(unit PRIVATE -Wall -Wextra -Wpedantic -Werror)
file(GLOB CORPUS_FILES CONFIGURE_DEPENDS generated_corpus_*.cpp)
add_executable(models generated_models.cpp ${CORPUS_FILES})
target_link_libraries(models PRIVATE Webmania::cpp Threads::Threads)
target_compile_options(models PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(lifetime lifetime.cpp)
target_link_libraries(lifetime PRIVATE Webmania::cpp Threads::Threads)
target_compile_options(lifetime PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(file-results file_results.cpp)
target_link_libraries(file-results PRIVATE Webmania::cpp Threads::Threads)
target_compile_options(file-results PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)

add_executable(native006 native006.cpp)
target_link_libraries(native006 PRIVATE Webmania::cpp Threads::Threads)
target_compile_options(native006 PRIVATE -Wall -Wextra -Wpedantic -Werror)
